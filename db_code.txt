-- CourseAggregatorDB schema for a React + Express course aggregator
-- Creates DB if not exists, switches to it, and defines necessary tables, indexes, and views.

-- 1) Create database
DROP DATABASE CourseAggregatorDB;
CREATE DATABASE CourseAggregatorDB
    DEFAULT CHARACTER SET utf8mb4
    DEFAULT COLLATE utf8mb4_unicode_ci;

USE CourseAggregatorDB;

-- 2) Reference tables (dimensions)

-- Sources (e.g., GitHub repos or platforms)
CREATE TABLE IF NOT EXISTS sources (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,                 -- repository/platform name
    url VARCHAR(1024) NOT NULL,                 -- base URL to source home
    UNIQUE KEY uq_sources_name (name)
) ENGINE=InnoDB;

-- Languages (content language)
CREATE TABLE IF NOT EXISTS languages (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(16) NOT NULL,                  -- e.g., 'en', 'el'
    name VARCHAR(64) NOT NULL,                  -- e.g., 'English', 'Greek'
    UNIQUE KEY uq_languages_code (code),
    UNIQUE KEY uq_languages_name (name)
) ENGINE=InnoDB;

-- Levels (beginner, intermediate, advanced)
CREATE TABLE IF NOT EXISTS levels (
    id TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(32) NOT NULL,                  -- e.g., 'beginner'
    display_name VARCHAR(64) NOT NULL,          -- e.g., 'Beginner'
    UNIQUE KEY uq_levels_code (code)
) ENGINE=InnoDB;

-- Categories (thematic areas)
CREATE TABLE IF NOT EXISTS categories (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(128) NOT NULL,                 -- e.g., 'Data Science', 'Web Development'
    slug VARCHAR(128) NOT NULL,                 -- URL-friendly identifier
    UNIQUE KEY uq_categories_name (name),
    UNIQUE KEY uq_categories_slug (slug)
) ENGINE=InnoDB;

-- 3) Main entity: courses
CREATE TABLE IF NOT EXISTS courses (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(512) NOT NULL,
    short_description TEXT,                     -- σύντομη περιγραφή
    full_description LONGTEXT,                  -- αναλυτική περιγραφή
    language_id INT UNSIGNED NOT NULL,          -- γλώσσα
    level_id TINYINT UNSIGNED NOT NULL,         -- επίπεδο
    source_id INT UNSIGNED NOT NULL,            -- πηγή (repository/platform)
    source_repo_name VARCHAR(255),              -- repository name
    source_repo_url VARCHAR(1024),              -- repository URL
    access_url VARCHAR(1024),                   -- Link εγγραφής/πρόσβασης
    last_updated_at DATETIME NULL,              -- Ημερομηνία τελευταίας ενημέρωσης (from source)
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_courses_language
        FOREIGN KEY (language_id) REFERENCES languages(id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_courses_level
        FOREIGN KEY (level_id) REFERENCES levels(id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_courses_source
        FOREIGN KEY (source_id) REFERENCES sources(id)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    INDEX idx_courses_title (title),
    INDEX idx_courses_last_updated (last_updated_at),
    INDEX idx_courses_source (source_id),
    INDEX idx_courses_language (language_id),
    INDEX idx_courses_level (level_id)
) ENGINE=InnoDB;

-- 5) Optional: similarity edges (for recommended/related courses, can be populated by Spark ML)
CREATE TABLE IF NOT EXISTS course_similarities (
    source_course_id BIGINT UNSIGNED NOT NULL,
    target_course_id BIGINT UNSIGNED NOT NULL,
    score DECIMAL(5,4) NOT NULL DEFAULT 0.0000, -- similarity score [0,1]
    PRIMARY KEY (source_course_id, target_course_id),
    CONSTRAINT fk_cs_source
        FOREIGN KEY (source_course_id) REFERENCES courses(id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_cs_target
        FOREIGN KEY (target_course_id) REFERENCES courses(id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    INDEX idx_cs_score (score)
) ENGINE=InnoDB;

-- 5b) Clustering assignments (from Spark KMeans)
CREATE TABLE IF NOT EXISTS course_clusters (
    id BIGINT UNSIGNED NOT NULL,
    cluster_id INT NOT NULL,
    top_terms TEXT NULL,
    PRIMARY KEY (id),
    CONSTRAINT fk_cc_clusters_course
        FOREIGN KEY (id) REFERENCES courses(id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    INDEX idx_cc_clusters_cluster (cluster_id)
) ENGINE=InnoDB;

CREATE TABLE IF NOT EXISTS course_cluster_terms (
    cluster_id INT NOT NULL PRIMARY KEY,
    top_terms TEXT NOT NULL
) ENGINE=InnoDB;

-- 6) Analytics views (optional but useful)

-- Count of courses per source
CREATE OR REPLACE VIEW vw_stats_courses_per_source AS
SELECT s.id AS source_id, s.name AS source_name, COUNT(c.id) AS course_count
FROM sources s
LEFT JOIN courses c ON c.source_id = s.id
GROUP BY s.id, s.name;

-- Count of courses per language
CREATE OR REPLACE VIEW vw_stats_courses_per_language AS
SELECT l.id AS language_id, l.name AS language_name, COUNT(c.id) AS course_count
FROM languages l
LEFT JOIN courses c ON c.language_id = l.id
GROUP BY l.id, l.name;
